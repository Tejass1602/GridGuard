cmake_minimum_required(VERSION 3.16)
project(gridguard_core C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(gridguard_core src/gg_plant.c src/gg_metrology.c src/gg_protection.c src/gg_relay_fsm.c src/gg_nilm.c src/gg_protocol.c)
target_include_directories(gridguard_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(gridguard_core PUBLIC m)
target_compile_options(gridguard_core PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE gridguard_core)
add_test(NAME core_acceptance COMMAND test_core)
